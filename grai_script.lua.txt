local function RunExternalScript(url, name)
    Rayfield:Notify({Title = "Đang tải...", Content = "Đang tải " .. name, Duration = 2})
    local success, result = pcall(function() return loadstring(game:HttpGet(url))() end)
    if not success then
        Rayfield:Notify({Title = "Lỗi!", Content = "Không thể chạy " .. name, Duration = 5})
        warn("Script Error ("..name.."): " .. tostring(result))
    end
end
