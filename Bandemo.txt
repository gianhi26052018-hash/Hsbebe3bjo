local block = script.Parent

-- RemoteEvent
local event = game.ReplicatedStorage:FindFirstChild("LuckyEvent")
if not event then
    event = Instance.new("RemoteEvent")
    event.Name = "LuckyEvent"
    event.Parent = game.ReplicatedStorage
end

-- Khi chạm block
block.Touched:Connect(function(hit)
    local player = game.Players:GetPlayerFromCharacter(hit.Parent)
    if player then
        event:FireClient(player)
    end
end)

-- Xử lý hiệu ứng
event.OnServerEvent:Connect(function(player, choice)
    local char = player.Character
    if not char then return end

    local humanoid = char:FindFirstChild("Humanoid")
    if not humanoid then return end

    if choice == "⚡ Speed" then
        humanoid.WalkSpeed = 50

    elseif choice == "🦘 Jump" then
        humanoid.JumpPower = 100

    elseif choice == "💥 Boom" then
        local explosion = Instance.new("Explosion")
        explosion.Position = char.PrimaryPart.Position
        explosion.Parent = workspace
    end
end)

        frame.Visible = false

        local corner = Instance.new("UICorner", frame)

        gradient.Color = ColorSequence.new{

            ColorSequenceKeypoint.new(1, Color3.fromRGB(170,0,255))

        title.Text = "🎁 LUCKY BLOCK"

        title.BackgroundTransparency = 1

        local function createButton(text, posY)

                btn.BackgroundColor3 = Color3.fromRGB(30,30,30)
